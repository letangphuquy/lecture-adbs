\newcommand{\kw}[1]{\textbf{#1}}                                 % SQL keyword
\newcommand{\add}[1]{\textcolor{vinuRed}{#1}}                     % added condition
\newcommand{\code}[1]{\colorbox{vinuNavy!7}{\ttfamily\begin{tabular}[c]{@{}l@{}}#1\end{tabular}}}
\newcommand{\goesto}{\textcolor{vinuRed}{\Large$\Rightarrow$}}

\begin{document}
\color{vinuNavy}\footnotesize
\setlength{\tabcolsep}{6pt}
\begin{tabular}{@{}c c c@{}}
  {\sffamily\itshape implicit join through transitivity} & & {\sffamily\itshape after the rewrite}\\[2pt]
  \code{\kw{SELECT} * \kw{FROM} R, S, T\\
        \ \kw{WHERE} R.A = S.B \kw{AND} S.B = T.C\\
        \ } &
  \goesto &
  \code{\kw{SELECT} * \kw{FROM} R, S, T\\
        \ \kw{WHERE} R.A = S.B \kw{AND} S.B = T.C\\
        \ \ \ \add{\kw{AND} R.A = T.C}}
\end{tabular}
