\begin{frame}{Example database \src{L15}{3}}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \textbf{Reserves}(\underline{sid, bid, day}, rname)
      \begin{itemize}\small
        \item 1000 pages, 100 tuples per page $\Rightarrow$ 100,000 tuples
        \item Each tuple is 40 bytes long
        \item 100 boats, each equally likely
      \end{itemize}
    \column{0.5\textwidth}
      \textbf{Sailors}(\underline{sid}, sname, rating, age)
      \begin{itemize}\small
        \item 500 pages, 80 tuples per page $\Rightarrow$ 40,000 tuples
        \item Each tuple is 50 bytes long
        \item 10 different ratings, each equally likely
      \end{itemize}
  \end{columns}
  \vspace{2mm}
  \begin{itemize}
    \item We have \hl{B = 5} buffer pages to use for joins
    \item Remember: \hl{just counting page I/Os} (CPU and writing the final output are ignored)
  \end{itemize}
  \begin{extrabox}[Derived numbers we will reuse]
    $\sel{\mathit{bid}=100}$ keeps $1/100$ of Reserves $\to$ 10 pages' worth \quad
    $\sel{\mathit{rating}>5}$ keeps $5/10$ of Sailors $\to$ 250 pages' worth \quad
    page size $\approx 100 \times 40 = 80 \times 50 = 4000$ B
  \end{extrabox}
\end{frame}

\begin{frame}{Reminder: notation and cost formulas from Week 4}
  \small
  Notation: $[X]$ = number of \textbf{pages} of $X$, $|X|$ = number of \textbf{tuples} of $X$;
  in a join the \textbf{left} input is the \textbf{outer}. (L12 writes $M=[\text{outer}]$, $m=|\text{outer}|$, $N=[\text{inner}]$.)
  \vspace{1mm}
  \begin{center}
  \begin{tabular}{l l l}
    \toprule
    \textbf{Operator} & \textbf{I/O cost} & \textbf{Source} \\
    \midrule
    Page nested loops join   & $M + M \cdot N$                          & \src{L12}{10} \\
    Block nested loops join  & $M + \lceil M/(B-2) \rceil \cdot N$       & \src{L12}{13} \\
    Index nested loops join  & $M + m \cdot (\text{cost to find matching inner tuples})$ & \src{L12}{15} \\
    External sort            & $2N \cdot \#\text{passes}$, \ $\#\text{passes} = 1 + \lceil \log_{B-1} \lceil N/B \rceil \rceil$ & \src{L11}{12} \\
    Sort-merge join          & sort(outer) + sort(inner) + $(M + N)$     & \src{L12}{32} \\
    Selection on the fly     & 0 extra I/O (but the scan below still reads every page) & \src{L13}{6} \\
    \bottomrule
  \end{tabular}
  \end{center}
\end{frame}

\subsection{Plans 1--3: selection pushdown}

\begin{frame}[fragile]{Query plan 1 \src{L15}{4}}
  \begin{columns}[c]
    \column{0.5\textwidth}
\begin{lstlisting}
SELECT S.sname
FROM   Reserves R, Sailors S
WHERE  R.sid = S.sid
  AND  R.bid = 100
  AND  S.rating > 5
\end{lstlisting}
      \vspace{2mm}
      Here's a reasonable query plan: one join, every selection and the projection applied
      \emph{on the fly} at the top.
    \column{0.42\textwidth}
      \planbox{56mm}{\PlanOne}
  \end{columns}
\end{frame}

\begin{frame}{Query plan 1 cost \src{L15}{5}}
  \begin{columns}[c]
    \column{0.55\textwidth}
      \begin{itemize}
        \fitem{(page NLJ)}{\src{L12}{10}}: $M + M\cdot N$, \\ here: $[S] + [S]\cdot[R]$
        \item Scan Sailors: \textbf{500 I/Os}
        \item For each page of Sailors \\ scan Reserves: \textbf{1000 I/Os}
        \item Total $= 500 + 500 \cdot 1000$\\ \phantom{Total} $=$ \cost{500,500 I/Os}
      \end{itemize}
    \column{0.42\textwidth}
      \planbox{56mm}{\PlanOne}
  \end{columns}
\end{frame}

\begin{frame}{Query plan 1 cost analysis \src{L15}{6}}
  \begin{columns}[c]
    \column{0.55\textwidth}
      \begin{itemize}
        \item Cost: \cost{500,500 I/Os}
        \item \textbf{By no means a terrible plan!}
        \item Misses several opportunities
        \begin{itemize}
          \item Selections could be ``pushed'' down
          \item No use of indexes
        \end{itemize}
        \item Goal of optimisation: find \alert{faster plans} that compute the \alert{same answer}
      \end{itemize}
    \column{0.42\textwidth}
      \planbox{56mm}{\PlanOne}
  \end{columns}
\end{frame}

\compareplans{Selection pushdown \src{L15}{7}}{\PlanOne}{500,500 I/Os}{\PlanTwo}{Cost?}{%
  \begin{itemize}
    \fitem{(PNLJ, outer filtered on the fly)}{\src{L12}{10}}: $[S] + [\sel{\mathit{rating}>5}(S)] \cdot [R]$ ---
      the outer is still read in full once, but only its \emph{surviving} tuples drive the inner loop.
  \end{itemize}}

\begin{frame}{Query plan 2 cost \src{L15}{8}}
  \begin{columns}[c]
    \column{0.55\textwidth}
      \begin{itemize}
        \item Scan Sailors: \textbf{500 I/Os}
        \item For each page of \emph{high-rated} Sailors\\ scan Reserves: \textbf{1000 I/Os}
        \item Total $= 500 + \;???\; \cdot 1000$
        \item Remember: 10 ratings, all equally likely\\
              Total $= 500 + (500/2)\cdot 1000 =$ \cost{250,500 I/Os}
      \end{itemize}
      \begin{notebox}
        The 250 ``pages'' are never written anywhere: they are page-sized groups of qualifying tuples streaming out of $\sigma$.
      \end{notebox}
    \column{0.42\textwidth}
      \planbox[hl]{56mm}{\PlanTwo}
  \end{columns}
\end{frame}

\compareplans{Decision 1 \src{L15}{9}}{\PlanOne}{500,500 I/Os}{\PlanTwo}{250,500 I/Os}{%
  \textbf{Idea:} select \emph{before} joining $\Rightarrow$ fewer tuples enter the join $\Rightarrow$ fewer inner scans.
  Pushing down on the \textbf{left (outer)} pays off because the outer size \emph{multiplies} the inner scan.}

\compareplans{More selection pushdown \src{L15}{10}}{\PlanTwo}{250,500 I/Os}{\PlanThree}{Cost?}{%
  Now also push $\sel{\mathit{bid}=100}$ below the join, onto the \textbf{right (inner)} input.}

\begin{frame}{Query plan 3 cost \src{L15}{11}}
  \begin{columns}[c]
    \column{0.55\textwidth}
      \begin{itemize}
        \item Scan Sailors: \textbf{500 I/Os}
        \item For each page of high-rated Sailors, read through the Reserves tuples that match
        \item Total $= 500 + 250 \cdot \;???$
        \item For each scan of Reserves, we filter \emph{on the fly}
        \item \alert{Problem:} this does not actually save any I/Os --- to find matching Reserves tuples
              we still scan all \textbf{1000} pages of Reserves each time
        \item Total $= 500 + 250 \cdot 1000 =$ \cost{250,500 I/Os}
      \end{itemize}
    \column{0.42\textwidth}
      \planbox[hl]{56mm}{\PlanThree}
  \end{columns}
\end{frame}

\compareplans{Decision 2 \src{L15}{12}}{\PlanTwo}{250,500 I/Os}{\PlanThree}{250,500 I/Os}{%
  A selection in the \textbf{inner loop} of a nested loops join \alert{does not save I/Os} (same as having it above):
  I/O counts \emph{pages read}, and the filter runs only \emph{after} each Reserves page is read.}

\begin{frame}{Query plan 7 cost, part 2 \src{L15}{27}}
  \small
  External sort with 5 buffers (merge $B-1 = 4$ runs at a time):
  \begin{itemize}
    \item Reserves, 10 pages: $1 + \lceil \log_4 \lceil 10/5 \rceil \rceil = 1 + \lceil \log_4 2 \rceil =$ \textbf{2 passes}
          \quad Pass 0 $= 10$ to write; \ Pass 1 $= 2 \times 10 = 20$ to read/write
    \item Sailors, 250 pages: $1 + \lceil \log_4 \lceil 250/5 \rceil \rceil = 1 + \lceil 2.82 \rceil =$ \textbf{4 passes}
          \quad Pass 0 $= 250$ to write; \ Passes 1--3 $= 2 \times 250 = 500$ each
    \item Total $=$ scan both $(1000 + 500)$ + sort Reserves $(10 + 20)$
          + sort Sailors $(250 + 3 \times 500)$ + merge $(260)$ $=$ \cost{3540 I/Os}
  \end{itemize}
  \begin{extrabox}[Pass 0 costs only a write --- why?]
    L11 charges $2N$ per pass (read + write). Here pass 0 receives its tuples \emph{pipelined} from $\sigma$
    (iterator model, L13), so its read is already paid for by the $1000 + 500$ table scans.
  \end{extrabox}
\end{frame}

\begin{frame}{Our explanation: why materialisation does not help here}
  \begin{columns}[T]
    \column{0.55\textwidth}
      \footnotesize
      Materialising an input $I$ into $T$ pays a \textbf{one-time} cost to make \textbf{repeated} reads cheaper.
      If the inner is scanned $p$ times:
      \vspace{-1mm}
      \[
        \underbrace{p \cdot [I]}_{\text{no materialisation}}
        \quad\text{vs.}\quad
        \underbrace{[I] + [T] + p \cdot [T]}_{\text{read once, write } T\text{, scan } T\ p \text{ times}}
      \]
      Worth it only if $(p - 1)\,[I] > (p + 1)\,[T]$.
      \begin{itemize}
        \item Plan 5: $p = 10$: \ $9 \cdot 500 > 11 \cdot 250$ \ \checkmark\ (saves)
        \item Plan 8: $p = 84$: \ $83 \cdot 1000 > 85 \cdot 10$ \ \checkmark\ (saves)
        \item Plan 9: the outer (1 page) fits in \textbf{one block} $\Rightarrow p = 1$ $\Rightarrow$ left side is 0:
              materialising can only \emph{add} $2[T]$ I/Os
      \end{itemize}
    \column{0.42\textwidth}
      \begin{extrabox}[Is it because of the algorithm?]
        \footnotesize No --- it is the \textbf{number of inner passes}. BNLJ only matters because it makes $p = 1$.
        A page NLJ with a 1-page outer would also scan the inner once, and materialisation would not help either.
      \end{extrabox}
      \begin{extrabox}[What about the outer side?]
        \footnotesize The outer is read exactly once anyway (pipelined into the block buffer), so there is nothing to save.
      \end{extrabox}
  \end{columns}
\end{frame}

\begin{frame}{Reminder: what we know about indexes (Weeks 3--4)}
  \small
  \begin{columns}[T]
    \column{0.5\textwidth}
      \textbf{B+-tree search}
      \begin{itemize}
        \item Root-to-leaf; high fanout $\Rightarrow$ height rarely exceeds 3 \src{L09}{13}
        \item Top levels often held in the buffer pool \src{L09}{16}
        \item Here: ``indexes fit in memory'' $\Rightarrow$ \hl{search = 0 I/O}; we pay only for \emph{data} pages
      \end{itemize}
      \textbf{Keys}
      \begin{itemize}
        \item $\mathit{sid}$ is the key of Sailors $\Rightarrow$ \textbf{at most one} match per lookup
      \end{itemize}
    \column{0.5\textwidth}
      \textbf{Clustered vs.\ unclustered} \src{L08}{36, 39}
      \begin{itemize}
        \item \textbf{Clustered}: data file (mostly) sorted by the search key $\Rightarrow$ matching records sit on
              \emph{consecutive} pages; cost = \#pages holding matches
        \item \textbf{Unclustered}: matches scattered; cost $\approx$ \#matching records (one I/O each)
        \item At most one clustered index per table
      \end{itemize}
      \textbf{INLJ} \src{L12}{15}
      \begin{itemize}
        \item $M + m \cdot (\text{search} + \text{matching pages / tuples})$
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Meta-lessons}
  \begin{enumerate}
    \item \textbf{Same query, same answer, $\sim$500$\times$ difference in cost} --- that is why optimisers exist.
    \item Gains came from all three kinds of plan-space moves: \textbf{logical rewrites} (pushdown, reordering),
          \textbf{physical choices} (PNLJ $\to$ SMJ $\to$ BNLJ $\to$ INLJ, scan $\to$ index scan) and
          \textbf{materialisation} decisions.
    \item \textbf{Moves interact}: a pushdown useless in Plan 3 became useful after materialisation (Plan 5);
          materialisation useful in Plan 8 was useless in Plan 9 $\Rightarrow$ we need a \emph{systematic search} with a \emph{cost model}.
    \item Heuristics are \textbf{usually} good, not always --- and every number rests on statistics and assumptions.
  \end{enumerate}
\end{frame}
